cmake_minimum_required(VERSION 3.24)
project(RemoteDeskNative VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RD_BUILD_MEDIA "Build the Windows GPU remote desktop worker" OFF)
option(RD_USE_QUINN "Use the Rust Quinn transport" ON)
option(RD_OPENSSL_TESTS "Build OpenSSL 3.5 transport tests when available" ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(native-tests tests/realtime_test.cpp)
target_include_directories(native-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(native-tests PRIVATE Threads::Threads)
add_test(NAME realtime-primitives COMMAND native-tests)
if(MSVC)
 target_compile_options(native-tests PRIVATE /W4 /WX /utf-8 /UNDEBUG)
else()
 target_compile_options(native-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
if(RD_OPENSSL_TESTS OR NOT RD_USE_QUINN)
 find_package(OpenSSL 3.5 QUIET)
 if(OpenSSL_FOUND)
  add_executable(quic-tests tests/quic_test.cpp)
  target_include_directories(quic-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(quic-tests PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  if(WIN32)
   target_link_libraries(quic-tests PRIVATE ws2_32)
  endif()
  add_test(NAME openssl-quic-udp COMMAND quic-tests)
 elseif(NOT RD_USE_QUINN AND RD_BUILD_MEDIA)
  message(FATAL_ERROR "The alternative transport requires OpenSSL 3.5 or newer")
 else()
  message(STATUS "OpenSSL QUIC tests skipped: OpenSSL >=3.5 was not found")
 endif()
endif()
if(WIN32)
 add_executable(capture-probe capture/capture_probe.cpp)
 target_compile_definitions(capture-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_compile_options(capture-probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_link_libraries(capture-probe PRIVATE d3d11 dxgi user32)
endif()
if(RD_BUILD_MEDIA)
 if(NOT WIN32)
  message(FATAL_ERROR "Desktop capture/rendering requires Windows 10/11")
 endif()
 find_path(FFMPEG_INCLUDE_DIR NAMES libavcodec/avcodec.h REQUIRED)
 foreach(component avcodec avutil swscale swresample)
  find_library(FFMPEG_${component}_LIB NAMES ${component} REQUIRED)
  list(APPEND FFMPEG_LIBS ${FFMPEG_${component}_LIB})
 endforeach()
 find_path(OPUS_INCLUDE_DIR NAMES opus/opus.h REQUIRED)
 find_library(OPUS_LIBRARY NAMES opus REQUIRED)
 add_executable(remote-media app/media_main.cpp)
 target_include_directories(remote-media PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${FFMPEG_INCLUDE_DIR} ${OPUS_INCLUDE_DIR})
 target_compile_definitions(remote-media PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_compile_options(remote-media PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4> $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_link_libraries(remote-media PRIVATE ${FFMPEG_LIBS} ${OPUS_LIBRARY} d3d11 dxgi user32 gdi32 ole32 avrt uuid Threads::Threads)
 if(RD_USE_QUINN)
  set(RD_QUINN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../target/release" CACHE PATH "Rust release directory")
  find_library(RD_QUINN_IMPORT NAMES remote_quic.dll remote_quic PATHS "${RD_QUINN_DIR}" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(remote-media PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../crates/remote-quic/include")
  target_compile_definitions(remote-media PRIVATE RD_USE_QUINN)
  target_link_libraries(remote-media PRIVATE "${RD_QUINN_IMPORT}")
  if(EXISTS "${RD_QUINN_DIR}/remote_quic.dll")
   add_custom_command(TARGET remote-media POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RD_QUINN_DIR}/remote_quic.dll" "$<TARGET_FILE_DIR:remote-media>")
  endif()
 else()
  target_link_libraries(remote-media PRIVATE OpenSSL::SSL OpenSSL::Crypto ws2_32)
 endif()
 install(TARGETS remote-media RUNTIME DESTINATION .)
endif()
